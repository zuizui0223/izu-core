"""65-node ecological weight admission on embedded archived joint states."""
from pathlib import Path
from itertools import combinations_with_replacement
import hashlib,json,time
import numpy as np
from scripts.audit_child_snapshot_probe import config,exposure,rounded,ecological_weights,atomic_json
from scripts.model3_weighted_factor_bound import bounded_weighted
c=Path(__file__).resolve().parents[1];root=c.parents[2];source=root/'outputs/model3_precision_feasibility/propagation1000';out=c/'weight65';out.mkdir(exist_ok=True)
if (out/'summary.json').exists():raise ValueError('preserve evidence')
axis=np.linspace(0,1,65);pairs=list(combinations_with_replacement(range(65),2));lookup={p:i for i,p in enumerate(pairs)}
positions=np.array([lookup[(a*8,b*8)] for a,b in combinations_with_replacement(range(9),2)])
rows=[]
for setting in ['assurance_cost','prior_selfing']:
 for arm in ['near','far']:
  for scheme in ['jump','heat_fv']:
   key=f'{setting}_{arm}_{scheme}';file=source/(key+'.npz');receipt=json.loads((source/(key+'.json')).read_text());assert hashlib.sha256(file.read_bytes()).hexdigest()==receipt['checkpoint_sha256']
   with np.load(file) as a:counts=a['baseline_200'].copy()
   state,_=rounded((counts.reshape(45,45,45),[np.eye(45)]*3));core,fs=state
   embedded=[]
   for f in fs:
    e=np.zeros((len(pairs),f.shape[1]));e[positions]=f;embedded.append(e)
   small=np.einsum('abc,ia,jb,kc->ijk',core,*fs,optimize=True)
   weights=ecological_weights(core,embedded,(axis,)*3,exposure(76001,arm).visitors[200],config(setting,.01))
   for kind in ['self','donor','recipient']:
    start=time.monotonic();row=dict(key=key,kind=kind,input_sha256=receipt['checkpoint_sha256'])
    try:
     xy=weights[kind+'_xy'];aw=np.broadcast_to(weights[kind+'_a'],(len(pairs),))
     if kind!='self':xy=xy[:,:,0]
     expected=small*xy[weights['maps'][0][positions,None],weights['maps'][1][None,positions]][:,:,None]*aw[positions][None,None,:]
     mass=float(expected.sum());assert mass>0
     (newcore,newfs),bound=bounded_weighted((core,embedded),xy,aw,weights['maps'],absolute_l1=mass*1e-8,budget=2_000_000)
     actual=np.einsum('abc,ia,jb,kc->ijk',newcore,*[f[positions] for f in newfs],optimize=True)
     totals=[np.abs(f).sum(axis=0) for f in newfs];inside=[np.abs(f[positions]).sum(axis=0) for f in newfs]
     # Sum disjoint first-outside-axis regions to avoid subtractive cancellation.
     mask=np.ones(len(pairs),bool);mask[positions]=False;outside=[np.abs(f[mask]).sum(axis=0) for f in newfs]
     outside_bound=sum(float(np.einsum('abc,a,b,c',np.abs(newcore),*[outside[j] if j==k else inside[j] if j<k else totals[j] for j in range(3)],optimize=True)) for k in range(3))
     error_bound=float(np.abs(actual-expected).sum())+outside_bound
     row.update(status='passed' if error_bound<=bound['absolute_l1_bound']+mass*1e-10 else 'failed',actual_l1_upper=error_bound,relative_l1_upper=error_bound/mass,mass=mass,bound=bound)
    except (ValueError,MemoryError,ArithmeticError) as e:row.update(status='not_evaluable',reason=str(e))
    row['seconds']=time.monotonic()-start;rows.append(row);atomic_json(out/(key+'_'+kind+'.json'),row);print(key,kind,row['status'],row.get('relative_l1_upper'),flush=True)
atomic_json(out/'summary.json',dict(scope='65-node weighting on embedded n9 states only, not high-grid trajectories',rows=rows))
